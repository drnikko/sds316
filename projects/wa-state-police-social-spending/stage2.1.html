Question 3 Part 2: In the largest and smallest counties in WA (by population), how does social spending per capita compare to law enforcement spending per capita?

#getting king and garfield and just looking at 2025
two_counties_social = fit_social_counties[fit_social_counties["Governments/$ Dollars"].isin(["King County", "Garfield County"])][["Governments/$ Dollars", "2025"]].copy()

#adding populations
populations = {"King County": 2269675, "Garfield County": 2416}

two_counties_social["Population"] = (
    two_counties_social["Governments/$ Dollars"].map(populations)
)

#social spending per capita
two_counties_social["Social_per_capita"] = (
    two_counties_social["2025"] / two_counties_social["Population"]
)
